VAR ltv =
    CALCULATE(SUM(cleaned_ecommerce_data1[revenue]),
        ALLEXCEPT(cleaned_ecommerce_data1,
            cleaned_ecommerce_data1[customer_unique_id]))

    IF(ltv >= 1000, "High Value",
        IF(ltv >= 300, "Mid Value", "Low Value"))
